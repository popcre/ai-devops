```bash
    map({conclusion:(.[0].conclusion // "null"),count:length})),

    select(.startedAt != null and .updatedAt != null) |
    ((.updatedAt|fromdateiso8601)-(.startedAt|fromdateiso8601))] |
```

```bash

  --json number,headRefOid > "$dir/open-prs.json"
tmp=$(mktemp)
while IFS= read -r n; do
  ./bin/ai-gh pr view "$n" --repo popcre/ai-devops --json files |
    jq -r --arg n "$n" '.files[].path | [$n,.] | @tsv'
done < <(jq -r '.[].number' "$dir/open-prs.json" | tr -d '\r') |
  sort -k2,2 -k1,1n > "$tmp"
mv "$tmp" "$dir/open-pr-files.tsv"
```

```bash

  awk '$1>1 {count=$1; $1=""; sub(/^ +/,""); print count "\t" $0}' |
  sort -t $'\t' -k1,1nr -k2,2 >> "$dir/open-pr-overlap.tsv"
```

There were 15 open pull requests, 88 pull-request/path rows, 69 unique paths,
and 15 paths present in more than one pull request. These are path overlaps,
